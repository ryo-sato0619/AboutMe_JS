Sub ExtractDataBasedOnKey() Dim keyValue As String Dim srcWS As Worksheet, destWS As Worksheet Dim srcRange As Range Dim srcData As Variant, tempList() As Variant, finalOutput() As Variant Dim lastRow As Long, i As Long, j As Long, outputRow As Long Dim numRows As Long Dim columnIndices As Variant ' キーとなる文字を、アクティブシートのV4から取得 keyValue = ActiveSheet.Range("V4").Value ' シートの指定（「情報集約用」からデータ取得、「グラフ用」に貼付け） Set srcWS = ThisWorkbook.Sheets("情報集約用") Set destWS = ThisWorkbook.Sheets("グラフ用") ' 「情報集約用」シートのC列の最終行を取得 (※C列が必ず値として使われる前提) lastRow = srcWS.Cells(srcWS.Rows.Count, "C").End(xlUp).Row ' 3行目から最終行までの範囲。今回はA~N列（14列）を一括取得する Set srcRange = srcWS.Range("A3:N" & lastRow) srcData = srcRange.Value ' srcDataは2次元配列 (1 to numRows, 1 to 14) ' 取得する列：A, B, C, D, E, F, G, N → 列番号 1, 2, 3, 4, 5, 6, 7, 14 columnIndices = Array(1, 2, 3, 4, 5, 6, 7, 14) ' 最大行数は取得範囲の行数 numRows = UBound(srcData, 1) ' 一時的な出力配列を初期化（最大 numRows 行分のデータがあると仮定） ReDim tempList(1 To numRows, 1 To UBound(columnIndices) + 1) outputRow = 0 ' 3行目が配列の1行目に相当するので、全行ループ ' キーワードは「情報集約用」シートのC列（配列の3列目）の値と比較 For i = 1 To numRows If CStr(srcData(i, 3)) = keyValue Then outputRow = outputRow + 1 For j = LBound(columnIndices) To UBound(columnIndices) tempList(outputRow, j + 1) = srcData(i, columnIndices(j)) Next j End If Next i If outputRow = 0 Then MsgBox "一致するデータが見つかりませんでした。", vbInformation, "完了" Exit Sub End If ' 出力配列を実際の抽出件数にリサイズ ReDim finalOutput(1 To outputRow, 1 To UBound(columnIndices) + 1) For i = 1 To outputRow For j = 1 To UBound(columnIndices) + 1 finalOutput(i, j) = tempList(i, j) Next j Next i ' シート「グラフ用」のA3セルを起点に結果を貼り付け destWS.Range("A3").Resize(outputRow, UBound(columnIndices) + 1).Value = finalOutput MsgBox outputRow & " 行のデータを抽出して貼り付けました。", vbInformation, "完了" End Sub
