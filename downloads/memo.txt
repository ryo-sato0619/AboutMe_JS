# Base64URLエンコード関数
function To-Base64Url($bytes) {
    [Convert]::ToBase64String($bytes).Replace('+','-').Replace('/','_').TrimEnd('=')
}

# JSON → UTF8 → Base64URL
$headerEncoded  = To-Base64Url([System.Text.Encoding]::UTF8.GetBytes((ConvertTo-Json $header -Compress)))
$payloadEncoded = To-Base64Url([System.Text.Encoding]::UTF8.GetBytes((ConvertTo-Json $payload -Compress)))
$jwtUnsigned    = "$headerEncoded.$payloadEncoded"

# digest生成（SHA-256）
$sha256      = [System.Security.Cryptography.SHA256]::Create()
$digestBytes = $sha256.ComputeHash([System.Text.Encoding]::UTF8.GetBytes($jwtUnsigned))

# Key Vault署名（RS256）
$kvUri       = "https://$keyVaultName.vault.azure.net"
$accessToken = (Get-AzAccessToken -ResourceUrl $kvUri).Token
$signUri     = "$kvUri/keys/$keyName/sign?api-version=7.4"

$body = @{
    alg   = "RS256"
    value = [Convert]::ToBase64String($digestBytes)
} | ConvertTo-Json

$response  = Invoke-RestMethod -Method Post -Uri $signUri -Headers @{ Authorization = "Bearer $accessToken" } -Body $body -ContentType "application/json"
$signature = $response.value

# JWT完成
$jwt = -join ($jwtUnsigned, ".", To-Base64Url([Convert]::FromBase64String($signature)))

# 保存
Set-Content -Path "./jwt.txt" -Value $jwt
Write-Host "✅ JWT生成完了：jwt.txt に保存しました"
