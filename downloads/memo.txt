Connect-MgGraph -Scopes "ServiceMessage.Read.All"

$messages = Get-MgServiceAnnouncementMessage -Top 50

# タイトルに "Teams" および "User" 関連のキーワードが含まれるものを抽出
$filtered = $messages | Where-Object {
    ($_.Title -match "Teams") -and
    ($_.Title -match "User|Impact|Access|Experience|Unable|Issue|Problem")
}

# 結果表示
$filtered | Select-Object Title, Severity, LastModifiedDateTime | Format-Table -AutoSize
